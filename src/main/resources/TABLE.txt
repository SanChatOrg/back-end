create table user (
    user_no bigint primary key not null auto_increment,
    user_name varchar(50) not null,
    user_birth date not null,
    user_id varchar(50) not null,
    user_pw varchar(255) not null ,
    created_at timestamp default current_timestamp,
    updated_at timestamp default current_timestamp,
    user_intro text
);

create table walk (
    walk_no bigint primary key auto_increment,
    walk_time time,
    walk_distance float,
    walk_date date
);

-- 채팅 테이블
drop table if exists chat_participant;
create table chat_participant (
    chat_part_no bigint primary key not null auto_increment,
    user_no bigint not null,
    chat_room_no bigint not null,
    joined_at timestamp default current_timestamp,
    unique(chat_room_no, user_no)
);

drop table if exists chat_room;
create table chat_room (
	chat_room_no bigint primary key not null auto_increment,
    chat_room_name varchar(50),
    created_at timestamp default current_timestamp,
    updated_at timestamp default current_timestamp,
    is_group_chat boolean default FALSE
);

drop table if exists chat_message;
create table chat_message (
	chat_message_no bigint primary key not null auto_increment,
    message_type enum('ENTER', 'CHAT', 'EXIT') not null,
    message_content text,
    sent_at timestamp default current_timestamp,
	user_no bigint not null,
    message_content_type enum('TEXT', 'IMAGE', 'VIDEO', 'FILE') not null,
    chat_room_no bigint not null
);

create table dog
(
    dog_no     bigint primary key auto_increment,
    dog_name  varchar(50)  not null,
    dog_breed varchar(50) not null,
    dog_gender char(1) not null,
    birth_date date         not null,
    is_neutered boolean,
    created_at timestamp default current_timestamp,
    updated_at timestamp default current_timestamp,
    user_no bigint not null
);

-- 커뮤니티 테이블
drop table if exists community;

create table community (
    community_no bigint primary key not null auto_increment,
    community_content text not null,
    file_path varchar(255) not null,
    created_at datetime default current_timestamp,
    updated_at datetime default current_timestamp on update current_timestamp,
    community_view_count bigint default 0,
    community_like_count bigint default 0,
    community_reply_count bigint default 0,
    community_deleted varchar(1) default 'n',
    file_public_id varchar(255),
    user_no bigint not null
);


-- 커뮤니티 좋아요
drop table if exists community_like;
create table community_like (
    like_no bigint primary key not null auto_increment,
    created_at timestamp null,
    is_liked varchar(1) not null default 'n',
    user_no bigint not null,
    community_no bigint not null
);

-- 커뮤니티 댓글
drop table if exists community_reply;
create table community_reply (
    reply_no bigint primary key not null auto_increment,
    reply_content text null,
    created_at timestamp null,
    reply_parent_no bigint null,
    community_no bigint not null
);

-- 해시태그
drop table if exists hashtag;
create table hashtag (
    hashtag_no bigint primary key not null auto_increment,
    hashtag_name varchar(255) null
);

CREATE TABLE Photo (
                       photo_no BIGINT AUTO_INCREMENT PRIMARY KEY, -- 사진 번호 (Primary Key)
                       photo_url VARCHAR(255) NOT NULL,           -- 사진 링크
                       photo_name VARCHAR(100),                   -- 사진명
                       photo_type VARCHAR(50),                    -- 사진 타입
                       created_at TIMESTAMP DEFAULT CURRENT_TIMESTAMP, -- 사진 등록일
                       related_type ENUM('USER', 'POST', 'CHAT', 'DOG') NOT NULL, -- 사진 연관 타입
                       related_entity_id BIGINT NOT NULL          -- 사진 연관 아이디
);
